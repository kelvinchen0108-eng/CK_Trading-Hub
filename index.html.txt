    <script>
        let db = JSON.parse(localStorage.getItem('ck_trading_v5')) || { workspaces: [], pfAccounts: [] };
        let currentWorkspaceId = null; let chart = null;
        function save() { localStorage.setItem('ck_trading_v5', JSON.stringify(db)); }
        function showPage(id) { document.querySelectorAll('.page-content').forEach(p => p.classList.remove('active')); document.getElementById(id).classList.add('active'); if(id === 'page-home') renderHome(); if(id === 'page-propfirm') renderPF(); }
        function goToHome() { currentWorkspaceId = null; document.getElementById('breadcrumb').classList.add('hidden'); showPage('page-home'); }
        
        // 刪除策略工作區
        function deleteWorkspace(id) {
            if(confirm('確定要永久刪除此交易策略工作區嗎？資料將無法恢復。')) {
                db.workspaces = db.workspaces.filter(ws => ws.id !== id);
                save(); renderHome();
            }
        }

        function createStrategy() {
            const name = document.getElementById('ns-name').value;
            const checklistRaw = document.getElementById('ns-checklist').value;
            if(!name) return;
            const checklist = checklistRaw ? checklistRaw.split(',').map(s => s.trim()).filter(s => s !== "") : [];
            const ws = { id: Date.now(), name: name, initialCapital: parseFloat(document.getElementById('ns-cap').value) || 10000, defaultRR: parseFloat(document.getElementById('ns-rr').value) || 2.0, riskPerTrade: parseFloat(document.getElementById('ns-risk').value) || 100, checklist: checklist, trades: [] };
            db.workspaces.push(ws); save(); renderHome();
        }

        // 需求：策略卡片補上勝率、RR 與刪除按鈕
        function renderHome() {
            const list = document.getElementById('strategy-list');
            list.innerHTML = db.workspaces.map(ws => {
                const trades = ws.trades;
                const pnl = trades.reduce((s,t) => s + t.pnl, 0);
                const wins = trades.filter(t => t.result === 1).length;
                const winRate = trades.length ? ((wins / trades.length) * 100).toFixed(0) : 0;
                const winTrades = trades.filter(t => t.result === 1);
                const avgRR = winTrades.length ? (winTrades.reduce((s,t) => s + t.rr, 0) / winTrades.length).toFixed(1) : "0.0";

                return `
                    <div class="card bg-[#1e293b] shadow-xl strategy-card transition-all border border-white/5 relative group" onclick="enterWorkspace(${ws.id})">
                        <button onclick="event.stopPropagation(); deleteWorkspace(${ws.id})" class="absolute top-2 right-2 btn btn-circle btn-xs btn-ghost text-error opacity-0 group-hover:opacity-100 transition-opacity">
                            ✕
                        </button>
                        <div class="card-body p-6">
                            <h2 class="card-title text-primary font-black italic tracking-tight">${ws.name}</h2>
                            <div class="grid grid-cols-2 gap-2 mt-2">
                                <div class="text-[10px] uppercase font-bold text-slate-500">Win Rate: <span class="text-white">${winRate}%</span></div>
                                <div class="text-[10px] uppercase font-bold text-slate-500">Avg RR: <span class="text-white">${avgRR}</span></div>
                            </div>
                            <div class="flex justify-between items-center mt-4 border-t border-white/10 pt-2">
                                <span class="text-xs font-bold text-slate-400">Balance: $${(ws.initialCapital + pnl).toFixed(0)}</span>
                                <span class="${pnl>=0?'text-success':'text-error'} font-black text-xl">${pnl>=0?'+':''}$${pnl.toFixed(0)}</span>
                            </div>
                        </div>
                    </div>`;
            }).join('');
        }

        function enterWorkspace(id) {
            currentWorkspaceId = id;
            const ws = db.workspaces.find(w => w.id === id);
            document.getElementById('current-nav-name').innerText = ws.name;
            document.getElementById('breadcrumb').classList.remove('hidden');
            document.getElementById('q-rr').value = ws.defaultRR;
            document.getElementById('q-date').value = new Date().toISOString().slice(0, 16);
            
            // 需求：檢核項打勾明顯化
            const container = document.getElementById('checklist-items-container');
            container.innerHTML = (ws.checklist && ws.checklist.length > 0) ? ws.checklist.map(item => `
                <label class="flex items-center gap-3 cursor-pointer opacity-90 hover:opacity-100 group">
                    <input type="checkbox" class="checkbox checkbox-success checkbox-sm rule-check" data-rule="${item}" />
                    <span class="transition-all">${item}</span>
                </label>`).join('') : '<p class="opacity-40 italic text-slate-400 text-[8px]">無檢核項</p>';

            initMonthFilter(ws.trades);
            showPage('page-detail'); updateDetail();
        }

        function initMonthFilter(trades) {
            const select = document.getElementById('filter-month');
            select.innerHTML = '<option value="all">所有月份數據</option>';
            const months = [...new Set(trades.map(t => t.date.slice(0, 7)))].sort().reverse();
            months.forEach(m => select.innerHTML += `<option value="${m}">${m}</option>`);
        }

        function resetEntryFields() {
            const ws = db.workspaces.find(w => w.id === currentWorkspaceId);
            document.getElementById('q-note').value = "";
            document.querySelectorAll('.rule-check').forEach(c => c.checked = false);
        }

        function quickLog(result) {
            const ws = db.workspaces.find(w => w.id === currentWorkspaceId);
            const rr = parseFloat(document.getElementById('q-rr').value) || ws.defaultRR;
            const date = document.getElementById('q-date').value;
            const rulesMet = Array.from(document.querySelectorAll('.rule-check:checked')).map(cb => cb.getAttribute('data-rule'));
            ws.trades.push({ id: Date.now(), date: date, result: result, rr: rr, pnl: result === 1 ? ws.riskPerTrade * rr : -ws.riskPerTrade, rules: rulesMet, note: document.getElementById('q-note').value });
            save(); resetEntryFields(); updateDetail();
        }

        function updateDetail() {
            const ws = db.workspaces.find(w => w.id === currentWorkspaceId);
            const filterVal = document.getElementById('filter-month').value;
            let filteredTrades = (filterVal === 'all') ? ws.trades : ws.trades.filter(t => t.date.startsWith(filterVal));
            const total = filteredTrades.length;
            const wins = filteredTrades.filter(t => t.result === 1).length;
            const winRate = total ? (wins/total) : 0;
            const winTrades = filteredTrades.filter(t => t.result === 1);
            const avgRR = winTrades.length ? (winTrades.reduce((s,t) => s + t.rr, 0) / winTrades.length).toFixed(1) : "0.0";
            const avgWin = wins ? (winTrades.reduce((s,t) => s + t.pnl, 0) / wins) : 0;
            const avgLoss = (total - wins) ? Math.abs(filteredTrades.filter(t => t.result === 0).reduce((s,t) => s + t.pnl, 0) / (total - wins)) : 0;
            const ev = (winRate * avgWin - (1 - winRate) * avgLoss).toFixed(2);
            let peak = ws.initialCapital, currentBal = ws.initialCapital, maxDD = 0;
            filteredTrades.forEach(t => { currentBal += t.pnl; if(currentBal > peak) peak = currentBal; let dd = (peak - currentBal) / peak; if(dd > maxDD) maxDD = dd; });
            document.getElementById('d-balance').innerText = `$${currentBal.toFixed(0)}`;
            document.getElementById('d-winrate').innerText = `${(winRate*100).toFixed(1)}%`;
            document.getElementById('d-avgrr').innerText = avgRR;
            document.getElementById('d-ev').innerText = `$${ev}`;
            document.getElementById('d-mdd').innerText = `${(maxDD*100).toFixed(1)}%`;
            document.getElementById('d-roi').innerText = `${((filteredTrades.reduce((s,t)=>s+t.pnl,0)/ws.initialCapital)*100).toFixed(1)}%`;
            document.getElementById('d-count').innerText = total;
            renderTable(filteredTrades, ws.initialCapital);
            renderChart(filteredTrades, ws.initialCapital);
        }

        function renderTable(trades, startCap) {
            let bal = startCap;
            document.getElementById('detail-table-body').innerHTML = trades.map((t, i) => {
                bal += t.pnl;
                return `<tr><td>${i+1}</td><td>${t.date.replace('T', ' ')}</td><td><div class="badge ${t.result===1?'badge-success':'badge-error'} badge-xs"></div></td><td class="font-bold text-white">${t.rr}</td><td class="${t.pnl>=0?'text-success':'text-error'}">$${t.pnl.toFixed(0)}</td><td class="text-white">$${bal.toFixed(0)}</td><td class="text-[10px] opacity-70">${t.rules.join('/')} | ${t.note}</td><td><button onclick="deleteTrade(${t.id})" class="btn btn-ghost btn-xs text-error">X</button></td></tr>`;
            }).reverse().join('');
        }

        function renderChart(trades, startCap) {
            const ctx = document.getElementById('detailChart').getContext('2d');
            let bal = startCap;
            const dataPts = [startCap];
            trades.forEach(t => { bal += t.pnl; dataPts.push(bal); });
            if(chart) chart.destroy();
            chart = new Chart(ctx, { type: 'line', data: { labels: dataPts.map((_, i) => i), datasets: [{ data: dataPts, borderColor: '#00d7ff', backgroundColor: 'rgba(0, 215, 255, 0.1)', fill: true, tension: 0.1, pointRadius: 1 }] }, options: { plugins: { legend: false }, scales: { y: { beginAtZero: false, grid: { color: 'rgba(255,255,255,0.05)' } } } } });
        }

        function deleteTrade(tid) { if(confirm('確定刪除？')){ const ws = db.workspaces.find(w => w.id === currentWorkspaceId); ws.trades = ws.trades.filter(t => t.id !== tid); save(); updateDetail(); } }
        
        // PF Logic (保持 5.4 版本修復後的狀態)
        function addPFAccount() { const company = document.getElementById('npf-company').value, cost = parseFloat(document.getElementById('npf-cost').value) || 0; db.pfAccounts.push({ id: Date.now(), company, cost, status: 'eval', payouts: [] }); save(); renderPF(); }
        function updatePFCost(id, n) { db.pfAccounts.find(a=>a.id===id).cost = parseFloat(n)||0; save(); renderPFStats(); }
        function updatePFStatus(id, s) { db.pfAccounts.find(a=>a.id===id).status = s; save(); renderPFStats(); }
        function renderPF() {
            const tbody = document.getElementById('pf-table-body');
            tbody.innerHTML = db.pfAccounts.map(acc => {
                const totalPayout = acc.payouts ? acc.payouts.reduce((s,p) => s+p.amount, 0) : 0;
                return `<tr><td class="font-black text-primary">${acc.company}</td><td><input type="number" value="${acc.cost}" onchange="updatePFCost(${acc.id}, this.value)" class="input input-xs input-bordered w-24 bg-slate-900"></td><td><select onchange="updatePFStatus(${acc.id}, this.value)" class="select select-xs select-bordered bg-slate-900"><option value="eval" ${acc.status==='eval'?'selected':''}>考核中</option><option value="passed" ${acc.status==='passed'?'selected':''}>已通過 (PA)</option><option value="payout" ${acc.status==='payout'?'selected':''}>已分潤</option><option value="failed" ${acc.status==='failed'?'selected':''}>考試失敗</option><option value="blown" ${acc.status==='blown'?'selected':''}>帳號爆倉</option></select></td><td class="text-success font-bold">+$${totalPayout}</td><td><button onclick="addPayout(${acc.id})" class="btn btn-xs btn-outline btn-success">Payout</button></td></tr>`;

        }
        function renderPFStats() {
            const cost = db.pfAccounts.reduce((s,a)=>s+a.cost,0), pay = db.pfAccounts.reduce((s,a)=>s+(a.payouts?a.payouts.reduce((sp,p)=>sp+p.amount,0):0),0);
            const success = db.pfAccounts.filter(a => a.status === 'passed' || a.status === 'payout' || a.status === 'blown').length;
            const fail = db.pfAccounts.filter(a => a.status === 'failed').length;
            const total = success + fail;
            const passRate = total ? (success / total * 100).toFixed(0) : 0;
            document.getElementById('pf-total-cost').innerText=`$${cost.toFixed(0)}`; document.getElementById('pf-total-payout').innerText=`$${pay.toFixed(0)}`; document.getElementById('pf-pass-rate').innerText=`${passRate}%`; document.getElementById('pf-net-pnl').innerText=`$${(pay-cost).toFixed(0)}`; document.getElementById('pf-net-pnl').className=`text-2xl font-black ${(pay-cost)>=0?'text-accent':'text-error'}`;
        }
